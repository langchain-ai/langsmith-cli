```go
package config

import (
	"bytes"
	"fmt"
	"os"
	"path/filepath"

	"github.com/BurntSushi/toml"
)

// DefaultConfigPath returns ~/.langsmith/config.toml.
func DefaultConfigPath() string {
	home, _ := os.UserHomeDir()
	return filepath.Join(home, ".langsmith", "config.toml")
}

// Profile represents a single named profile.
type Profile struct {
	APIKey      string `toml:"api_key"`
	APIURL      string `toml:"api_url,omitempty"`
	WorkspaceID string `toml:"workspace_id,omitempty"`
}

// Config represents the full config file.
type Config struct {
	CurrentProfile string             `toml:"current_profile,omitempty"`
	Profiles       map[string]Profile `toml:"-"`
}

// tomlFile is the on-disk representation. Top-level keys are mixed with profile sections,
// so we decode in two passes: first the top-level scalars, then each remaining key as a Profile.
type tomlFile struct {
	CurrentProfile string `toml:"current_profile"`
}

// Load reads the config from the default path.
func Load() (*Config, error) {
	return LoadFrom(DefaultConfigPath())
}

// LoadFrom reads the config from the given path. Returns an empty Config if the file doesn't exist.
func LoadFrom(path string) (*Config, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		if os.IsNotExist(err) {
			return &Config{Profiles: map[string]Profile{}}, nil
		}
		return nil, fmt.Errorf("reading config: %w", err)
	}

	// First pass: decode top-level scalars.
	var top tomlFile
	md, err := toml.Decode(string(data), &top)
	if err != nil {
		return nil, fmt.Errorf("parsing config: %w", err)
	}

	cfg := &Config{
		CurrentProfile: top.CurrentProfile,
		Profiles:       make(map[string]Profile),
	}

	// Second pass: each undecoded top-level key is a profile section.
	for _, key := range md.Undecoded() {
		parts := key.String()
		// Only top-level keys (no dots) are profile names.
		if len(key) != 1 {
			continue
		}
		name := parts
		var p Profile
		// Re-decode just this section.
		type wrapper struct {
			Profile Profile `toml:"-"`
		}
		// Use a map approach: decode entire file as map, extract the section.
		var raw map[string]any
		if _, err := toml.Decode(string(data), &raw); err != nil {
			return nil, fmt.Errorf("parsing config profiles: %w", err)
		}
		section, ok := raw[name]
		if !ok {
			continue
		}
		sectionMap, ok := section.(map[string]any)
		if !ok {
			continue
		}
		if v, ok := sectionMap["api_key"].(string); ok {
			p.APIKey = v
		}
		if v, ok := sectionMap["api_url"].(string); ok {
			p.APIURL = v
		}
		if v, ok := sectionMap["workspace_id"].(string); ok {
			p.WorkspaceID = v
		}
		cfg.Profiles[name] = p
	}

	return cfg, nil
}

// SaveTo writes the config to the given path, creating parent directories as needed.
// The file is written with 0600 permissions since it contains API keys.
func (c *Config) SaveTo(path string) error {
	if err := os.MkdirAll(filepath.Dir(path), 0700); err != nil {
		return fmt.Errorf("creating config directory: %w", err)
	}

	var buf bytes.Buffer
	if c.CurrentProfile != "" {
		fmt.Fprintf(&buf, "current_profile = %q\n", c.CurrentProfile)
	}
	for name, p := range c.Profiles {
		buf.WriteString("\n")
		fmt.Fprintf(&buf, "[%s]\n", name)
		fmt.Fprintf(&buf, "api_key = %q\n", p.APIKey)
		if p.APIURL != "" {
			fmt.Fprintf(&buf, "api_url = %q\n", p.APIURL)
		}
		if p.WorkspaceID != "" {
			fmt.Fprintf(&buf, "workspace_id = %q\n", p.WorkspaceID)
		}
	}

	return os.WriteFile(path, buf.Bytes(), 0600)
}

// Save writes the config to the default path.
func (c *Config) Save() error {
	return c.SaveTo(DefaultConfigPath())
}

// ResolveProfile returns the active profile based on precedence:
// 1. flagProfile (from --profile)
// 2. envProfile (from LANGSMITH_PROFILE)
// 3. current_profile from config
// 4. "default" profile
// Returns nil if no matching profile is found.
func (c *Config) ResolveProfile(flagProfile, envProfile string) *Profile {
	name := flagProfile
	if name == "" {
		name = envProfile
	}
	if name == "" {
		name = c.CurrentProfile
	}
	if name == "" {
		name = "default"
	}
	if p, ok := c.Profiles[name]; ok {
		return &p
	}
	return nil
}

// MaskAPIKey masks an API key for display, showing first 8 and last 4 characters.
func MaskAPIKey(key string) string {
	if len(key) <= 12 {
		if len(key) <= 4 {
			return "****"
		}
		return key[:4] + "..." + key[len(key)-3:]
	}
	return key[:8] + "..." + key[len(key)-4:]
}
```

Note: The `LoadFrom` implementation above has a subtle inefficiency — it decodes the file twice. Here's a cleaner approach that should replace the second pass. Actually, let's simplify by decoding the whole file as a `map[string]any` once and extracting both the top-level key and profile sections in one pass:

Replace the `LoadFrom` function body with:

```go
func LoadFrom(path string) (*Config, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		if os.IsNotExist(err) {
			return &Config{Profiles: map[string]Profile{}}, nil
		}
		return nil, fmt.Errorf("reading config: %w", err)
	}

	var raw map[string]any
	if _, err := toml.Decode(string(data), &raw); err != nil {
		return nil, fmt.Errorf("parsing config: %w", err)
	}

	cfg := &Config{
		Profiles: make(map[string]Profile),
	}

	if v, ok := raw["current_profile"].(string); ok {
		cfg.CurrentProfile = v
	}

	for key, val := range raw {
		if key == "current_profile" {
			continue
		}
		section, ok := val.(map[string]any)
		if !ok {
			continue
		}
		var p Profile
		if v, ok := section["api_key"].(string); ok {
			p.APIKey = v
		}
		if v, ok := section["api_url"].(string); ok {
			p.APIURL = v
		}
		if v, ok := section["workspace_id"].(string); ok {
			p.WorkspaceID = v
		}
		cfg.Profiles[key] = p
	}

	return cfg, nil
}
```

```go
package cmd

import (
	"bufio"
	"context"
	"encoding/json"
	"fmt"

	"sort"
	"strings"

	"github.com/langchain-ai/langsmith-cli/internal/config"

// getConfigPath returns the config file path (from env override or default).
func getConfigPath() string {
	if v := os.Getenv("LANGSMITH_CONFIG_FILE"); v != "" {
		return v
	}
	return config.DefaultConfigPath()
}

func newProfileCmd() *cobra.Command {
	cmd := &cobra.Command{
		Use:   "profile",
		Short: "Manage LangSmith CLI profiles",
		Long: `Manage named profiles for switching between LangSmith configurations.

Profiles store API keys, endpoints, and workspace IDs in ~/.langsmith/config.toml.
Select a profile with --profile flag, LANGSMITH_PROFILE env var, or 'profile use'.

Examples:
  langsmith profile create prod --api-key lsv2_pt_... --api-url https://api.smith.langchain.com
  langsmith profile list
  langsmith profile use prod
  langsmith profile show prod
  langsmith profile delete staging`,
	}

	cmd.AddCommand(newProfileListCmd())
	cmd.AddCommand(newProfileShowCmd())

func newProfileCreateCmd() *cobra.Command {
	var (
		apiKey      string
		apiURL      string
		workspaceID string
	)

	cmd := &cobra.Command{
		Use:   "create <name>",
		Short: "Create a new profile",
		Args:  cobra.ExactArgs(1),
		RunE: func(cmd *cobra.Command, args []string) error {
			name := args[0]
			path := getConfigPath()

			cfg, err := config.LoadFrom(path)
			if err != nil {
				return err
			}

			if _, exists := cfg.Profiles[name]; exists {
				return fmt.Errorf("profile already exists: %s", name)
			}

			// Interactive prompts for missing required fields
			if apiKey == "" {
				apiKey = promptInput("API Key: ")
			}
			if apiKey == "" {
				return fmt.Errorf("api_key is required")
			}
			if apiURL == "" {
				apiURL = promptInputDefault("API URL", "https://api.smith.langchain.com")
			}
			if workspaceID == "" {
				workspaceID = promptInput("Workspace ID (optional, press enter to skip): ")
			}

			cfg.Profiles[name] = config.Profile{
				APIKey:      apiKey,
				APIURL:      apiURL,
				WorkspaceID: workspaceID,
			}

			// First profile becomes current automatically
			if len(cfg.Profiles) == 1 && cfg.CurrentProfile == "" {
				cfg.CurrentProfile = name
			}

			if err := cfg.SaveTo(path); err != nil {
				return err
			}

			result := map[string]string{"status": "created", "profile": name}
			data, _ := json.MarshalIndent(result, "", "  ")
			fmt.Fprintln(cmd.OutOrStdout(), string(data))
			return nil
		},
	}

	cmd.Flags().StringVar(&apiKey, "api-key", "", "LangSmith API key")
	cmd.Flags().StringVar(&apiURL, "api-url", "", "LangSmith API URL")
	cmd.Flags().StringVar(&workspaceID, "workspace-id", "", "LangSmith workspace ID")

	return cmd
}

// promptInput reads a line from stdin with the given prompt.
func promptInput(prompt string) string {
	fmt.Fprint(os.Stderr, prompt)
	scanner := bufio.NewScanner(os.Stdin)
	if scanner.Scan() {
		return strings.TrimSpace(scanner.Text())
	}
	return ""
}

// promptInputDefault reads a line with a default value shown.
func promptInputDefault(label, defaultVal string) string {
	fmt.Fprintf(os.Stderr, "%s [%s]: ", label, defaultVal)
	scanner := bufio.NewScanner(os.Stdin)
	if scanner.Scan() {
		v := strings.TrimSpace(scanner.Text())
		if v != "" {
			return v
		}
	}
	return defaultVal
}
```
